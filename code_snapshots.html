71# --- CITIZEN ASSISTANCE REQUEST DISPATCH ENDPOINT ---
72@app.route('/api/requests', methods=['GET', 'POST'])
73def handle_requests():
74    conn = get_db()
75    if request.method == 'POST':
76        data = request.json
77        conn.execute(
78            "INSERT INTO requests (email, location, req_type, status) VALUES (?, ?, ?, 'Pending')", 
79            (data.get('email', 'anonymous'), data.get('location'), data.get('req_type'))
80        )
81        conn.commit(); conn.close()
82        return jsonify({"success": True, "message": "Request saved to database."})
83    
84    reqs = conn.execute("SELECT * FROM requests ORDER BY id DESC").fetchall()
85    conn.close()
86    return jsonify([dict(r) for r in reqs])

88@app.route('/api/requests/<int:req_id>/accept', methods=['POST'])
89def accept_request(req_id):
90    conn = get_db()
91    conn.execute("UPDATE requests SET status='Dispatched' WHERE id=?", (req_id,))
92    conn.commit(); conn.close()
93    return jsonify({"success": True, "message": "Request accepted and dispatched."})
